## 7. 可直接复跑的命令

```powershell
$PY = ".venv\Scripts\python.exe"
$env:PYTHONPATH = "e:\Project\inv-trend-trading\src"

# 契约与基准
& $PY -m pytest "tests/test_数据流水线契约_v1.py" -q --no-header

    --junit-after .tmp/pipeline/after_junit.xml --out .tmp/pipeline/junit_对比.json

# data-pipeline 产物（两侧同一 fixture 路径 + 同一 run-id）
& $PY scripts\结构对比_产物_v1.py --surface data-pipeline --out .tmp/pipeline/fixed-after `
    --processed-dir processed_data --run-id a1b2c3d4e5f678901234abcd

# 制品对比（G3/G8，已闭环）
& $PY scripts\结构对比_v1.py --before .tmp/pipeline/fixed-before `
    --after .tmp/pipeline/fixed-after --out .tmp/pipeline/fixed_对比结果.json

# 静态与引用审计
```
